import re
from collections import Counter
docs = context.split("=== DOCUMENT BREAK ===")
def all_orgs(doc):
    subj = re.findall(r"\[ENT-\d+\]\s+(.+?)\n", doc)
    disp = re.findall(r"custody passed from the (.+?) to the (.+?)", doc)
    names = list(subj)
    for a, b in disp:
        names.append(a)
        names.append(b)
    return names

sets = []
for d in docs:
    names = all_orgs(d)
    sets.append(set(names))
    print(f"doc: {len(names)} mentions, {len(set(names))} unique")

inter = sets[0] & sets[1] & sets[2]
print("intersection:", inter)
